function prosesSingleItem(rawItemText, baseQty) { let text = rawItemText.trim(); if (!text) return; let itemQty = baseQty; let matchQty = text.match(/^(\d+)x\s*(.*)/i); if (matchQty) { itemQty = baseQty * parseInt(matchQty[1]); text = matchQty[2].trim(); } // 1. Ekstraksi Paket / Tanda Kurung (...) let parenMatch = text.match(/^(.*?)\((.*?)\)/); if (parenMatch) { let insideBracket = parenMatch[2].trim(); let subParts = insideBracket.split('+'); subParts.forEach(subPart => { let subText = subPart.trim(); // PERBAIKAN: Tangkap angka di dalam kurung walaupun tanpa huruf 'x' (contoh: "2 Manis CKW") let subQtyMatch = subText.match(/^(\d+)\s*(.*)/); let currentSubQty = 1; let cleanSubText = subText; if (subQtyMatch) { currentSubQty = parseInt(subQtyMatch[1]); cleanSubText = subQtyMatch[2].trim(); } // Proses item di dalam kurung dengan mengalikan kuantitas paket prosesSingleItem(cleanSubText, itemQty * currentSubQty); }); return; } // 2. Klasifikasi Kategori Terkunci let textLower = text.toLowerCase(); let kategori = ""; if (textLower.includes("telur")) { kategori = "Martabak Telur"; } else if (textLower.includes("tipker")) { kategori = "Tipker"; } else if (textLower.includes("qr manis") || textLower.includes("qr ") || textLower.includes("reguler")) { kategori = "Manis Reguler"; } else if (textLower.includes("cs ") || textLower.includes("manis mini") || textLower.includes("mini") || textLower.includes("ckw")) { kategori = "Manis Mini"; } else { kategori = "Manis Mini"; // Default fallback } tambahPenjualan(kategori, itemQty); ekstrakToppingItem(text, kategori, itemQty); }
